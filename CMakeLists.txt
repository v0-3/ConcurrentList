cmake_minimum_required(VERSION 3.16)
project(ConcurrentList LANGUAGES CXX)

include(CTest)
if(BUILD_TESTING)
    find_package(Threads REQUIRED)
    add_executable(concurrent_list_tests tests/concurrent_list_tests.cpp)
    add_executable(allocation_tests tests/allocation_tests.cpp)
    foreach(target concurrent_list_tests allocation_tests)
        target_include_directories(${target} PRIVATE src)
        target_compile_features(${target} PRIVATE cxx_std_11)
        set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
        target_link_libraries(${target} PRIVATE Threads::Threads)
        if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
            target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
        endif()
    endforeach()

    # Compile and instantiate each header without relying on other includes.
    foreach(header AtomicMarkableReference CoarseGrainedList FineGrainedList LazyList LockFreeList OptimisticList)
        set(source "${CMAKE_CURRENT_BINARY_DIR}/check_${header}.cpp")
        file(WRITE "${source}" "#include \"${header}.hpp\"\ntemplate class ${header}<int>;\n")
        target_sources(concurrent_list_tests PRIVATE "${source}")
    endforeach()

    foreach(group atomic coarse fine optimistic lazy lock_free)
        add_test(NAME ${group} COMMAND concurrent_list_tests ${group})
        set_tests_properties(${group} PROPERTIES TIMEOUT 120)
    endforeach()
    add_test(NAME allocations COMMAND allocation_tests)
    set_tests_properties(allocations PROPERTIES TIMEOUT 120)
endif()
